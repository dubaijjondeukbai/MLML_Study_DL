        "def fit_with_padding(frame, target_w=1280, target_h=720):\n",
        "    h, w = frame.shape[:2]\n",
        "\n",
        "    scale = min(target_w / w, target_h / h)\n",
        "\n",
        "    new_w = int(w * scale)\n",
        "    new_h = int(h * scale)\n",
        "\n",
        "    resized = cv2.resize(\n",
        "        frame,\n",
        "        (new_w, new_h),\n",
        "        interpolation=cv2.INTER_AREA\n",
        "    )\n",
        "\n",
        "    canvas = np.zeros(\n",
        "        (target_h, target_w, 3),\n",
        "        dtype=np.uint8\n",
        "    )\n",
        "\n",
        "    x = (target_w - new_w) // 2\n",
        "    y = (target_h - new_h) // 2\n",
        "\n",
        "    canvas[y:y + new_h, x:x + new_w] = resized\n",
        "\n",
        "    return canvas"
